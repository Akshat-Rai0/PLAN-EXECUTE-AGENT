1. PROJECT OVERVIEW
================================================================================

This implementation guide provides a complete blueprint for implementing a
multi-purpose browser automation agent using browser-use library with the
following capabilities:

1. FORM FILLING - Automated form submission on any website
2. BOOKING SYSTEM - Flight/hotel/train/movie ticket booking with user interaction
3. COMPARISON - Browse and compare products/services across multiple sites

The architecture is modular, allowing you to use components independently or
combined as needed.

from browser_use import Agent, Browser, ChatBrowserUse, ChatOpenAI, ChatAnthropic, ChatGoogle
from browser_use.tools.service import Tools
from config.settings import settings
from typing import Any

class BaseAgentFactory:
    """Factory for creating configured agents with consistent settings"""
    
    @staticmethod
    def get_llm():
        """Initialize LLM based on configuration"""
        if settings.llm_provider == "browser_use":
            return ChatBrowserUse(model=settings.llm_model)
        elif settings.llm_provider == "openai":
            return ChatOpenAI(model=settings.llm_model)
        elif settings.llm_provider == "anthropic":
            return ChatAnthropic(model=settings.llm_model)
        elif settings.llm_provider == "google":
            return ChatGoogle(model=settings.llm_model)
        else:
            return ChatBrowserUse(model="bu-2-0")
    
    @staticmethod
    def create_agent(
        task: str,
        tools: Tools | None = None,
        **kwargs
    ) -> Agent:
        """Create a configured agent with standard settings"""
        llm = BaseAgentFactory.get_llm()
        
        # Base configuration
        agent_config = {
            "task": task,
            "llm": llm,
            "max_steps": settings.max_steps,
            "max_failures": settings.max_failures,
            "use_vision": settings.use_vision,
            "llm_timeout": settings.llm_timeout,
            "step_timeout": settings.step_timeout,
            **kwargs
        }
        
        # Add tools if provided
        if tools:
            agent_config["tools"] = tools
        
        return Agent(**agent_config)
    
    @staticmethod
    def create_browser_agent(
        task: str,
        tools: Tools | None = None,
        headless: bool | None = None,
        use_cloud: bool | None = None,
        **kwargs
    ) -> Agent:
        """Create agent with browser configuration"""
        browser_config = {}
        
        if headless is not None:
            browser_config["headless"] = headless
        elif settings.headless:
            browser_config["headless"] = settings.headless
            
        if use_cloud is not None:
            browser_config["use_cloud"] = use_cloud
        elif settings.use_cloud:
            browser_config["use_cloud"] = settings.use_cloud
        
        return BaseAgentFactory.create_agent(
            task=task,
            tools=tools,
            **browser_config,
            **kwargs
        )

from browser_use import ActionResult, Tools
from browser_use.tools.service import Tools
from typing import Callable, Awaitable
import asyncio
import logging

logger = logging.getLogger(__name__)

class UserInteractionManager:
    """Manages different methods of user interaction"""
    
    def __init__(self, input_method: str = "cli"):
        self.input_method = input_method
        self.input_callback: Callable[[str], Awaitable[str]] | None = None
        self.history: list[dict] = []
    
    def set_input_callback(self, callback: Callable[[str], Awaitable[str]]):
        """Set custom input callback for web/API interfaces"""
        self.input_callback = callback
        logger.info(f"Custom input callback set for {self.input_method} interface")
    
    async def get_user_input(self, question: str) -> str:
        """Get user input using configured method"""
        try:
            if self.input_callback:
                response = await self.input_callback(question)
            elif self.input_method == "cli":
                response = input(f"{question}: ")
            elif self.input_method == "mock":
                response = "mock_response"
            elif self.input_method == "web":
                raise NotImplementedError("Web input requires callback implementation")
            else:
                raise ValueError(f"Unknown input method: {self.input_method}")
            
            # Log interaction
            self.history.append({
                "question": question,
                "response": response,
                "method": self.input_method,
                "timestamp": asyncio.get_event_loop().time()
            })
            
            return response
        except Exception as e:
            logger.error(f"Error getting user input: {e}")
            raise

# Global user interaction manager
user_manager = UserInteractionManager()

def get_user_tools() -> Tools:
    """Create user interaction tools for agent use"""
    tools = Tools()
    
    @tools.registry.action('Ask user a question and wait for their response')
    async def ask_user(question: str) -> ActionResult:
        """Ask the user a question and get their input"""
        try:
            response = await user_manager.get_user_input(question)
            return ActionResult(
                extracted_content=response,
                include_in_memory=True
            )
        except Exception as e:
            return ActionResult(error=f"Failed to get user input: {str(e)}")
    
    @tools.registry.action('Ask user for confirmation (yes/no)')
    async def ask_confirmation(question: str) -> ActionResult:
        """Ask user a yes/no question"""
        try:
            response = await user_manager.get_user_input(f"{question} (yes/no): ")
            is_confirmed = response.lower() in ['yes', 'y', 'true', '1']
            return ActionResult(
                extracted_content=str(is_confirmed),
                include_in_memory=True
            )
        except Exception as e:
            return ActionResult(error=f"Failed to get confirmation: {str(e)}")
    
    @tools.registry.action('Ask user to choose from options')
    async def ask_choice(question: str, options: list[str]) -> ActionResult:
        """Ask user to choose from provided options"""
        try:
            options_text = "\n".join([f"{i+1}. {opt}" for i, opt in enumerate(options)])
            response = await user_manager.get_user_input(f"{question}\n{options_text}\nEnter choice number: ")
            
            try:
                choice_index = int(response) - 1
                if 0 <= choice_index < len(options):
                    return ActionResult(
                        extracted_content=options[choice_index],
                        include_in_memory=True
                    )
                else:
                    return ActionResult(error="Invalid choice number")
            except ValueError:
                return ActionResult(error="Please enter a valid number")
        except Exception as e:
            return ActionResult(error=f"Failed to get user choice: {str(e)}")
    
    @tools.registry.action('Get numeric input from user')
    async def get_numeric_input(question: str, min_value: int | None = None, max_value: int | None = None) -> ActionResult:
        """Get numeric input from user with optional validation"""
        try:
            response = await user_manager.get_user_input(question)
            value = float(response)
            
            if min_value is not None and value < min_value:
                return ActionResult(error=f"Value must be at least {min_value}")
            if max_value is not None and value > max_value:
                return ActionResult(error=f"Value must be at most {max_value}")
            
            return ActionResult(
                extracted_content=str(value),
                include_in_memory=True
            )
        except ValueError:
            return ActionResult(error="Please enter a valid number")
        except Exception as e:
            return ActionResult(error=f"Failed to get numeric input: {str(e)}")
    
    return tools

from browser_use import ActionResult, Tools
from agents.base_agent import BaseAgentFactory
from typing import dict
import logging

logger = logging.getLogger(__name__)

class FormAgent:
    """Specialized agent for form filling tasks"""
    
    def __init__(self):
        self.tools = self._get_form_tools()
    
    def _get_form_tools(self) -> Tools:
        """Create form-specific tools"""
        tools = Tools()
        
        @tools.registry.action('Prepare form data for filling')
        async def prepare_form_data(form_data: dict[str, str]) -> ActionResult:
            """Prepare and validate form data"""
            # Validate that form data is not empty
            if not form_data:
                return ActionResult(error="Form data cannot be empty")
            
            formatted_data = "\n".join([f"- {key}: {value}" for key, value in form_data.items()])
            logger.info(f"Form data prepared: {formatted_data}")
            
            return ActionResult(
                extracted_content=f"Form data prepared:\n{formatted_data}",
                include_in_memory=True
            )
        
        @tools.registry.action('Validate form fields before submission')
        async def validate_form_fields(validation_rules: dict[str, str]) -> ActionResult:
            """Validate form data against provided rules"""
            # Simple validation logic
            errors = []
            
            for field, rule in validation_rules.items():
                if rule == "required" and not field:
                    errors.append(f"{field} is required")
                elif rule == "email" and "@" not in field:
                    errors.append(f"{field} must be a valid email")
            
            if errors:
                return ActionResult(error=f"Validation failed: {', '.join(errors)}")
            
            return ActionResult(
                extracted_content="Form validation passed",
                include_in_memory=True
            )
        
        return tools
    
    async def fill_form_task(
        self,
        url: str,
        form_data: dict[str, str],
        submit: bool = True,
        validation_rules: dict[str, str] | None = None
    ) -> str:
        """Execute form filling task"""
        task = f"""
        Go to {url}
        Fill out the form with the following data:
        {self._format_form_data(form_data)}
        """
        
        if validation_rules:
            task += f"\nValidate the form using these rules: {validation_rules}"
        
        if submit:
            task += "\nThen submit the form and report the result."
        else:
            task += "\nDo not submit the form, just fill it out and show me the filled form."
        
        logger.info(f"Starting form filling task for {url}")
        
        agent = BaseAgentFactory.create_browser_agent(
            task=task,
            tools=self.tools
        )
        
        try:
            history = await agent.run()
            result = history.final_result()
            logger.info(f"Form filling completed: {result}")
            return result
        except Exception as e:
            logger.error(f"Form filling failed: {e}")
            raise
    
    def _format_form_data(self, form_data: dict[str, str]) -> str:
        """Format form data for task description"""
        return "\n".join([f"- {key}: {value}" for key, value in form_data.items()])
    
    async def fill_form_with_user_input(self, url: str) -> str:
        """Fill form by collecting data from user interactively"""
        from tools.user_interaction import user_manager
        
        form_data = {}
        
        print("Enter form field data (empty field name to finish):")
        
        while True:
            field_name = await user_manager.get_user_input("Field name: ")
            if not field_name:
                break
            
            field_value = await user_manager.get_user_input(f"Value for {field_name}: ")
            form_data[field_name] = field_value
        
        submit_response = await user_manager.get_user_input("Submit form? (yes/no): ")
        submit = submit_response.lower() in ['yes', 'y']
        
        return await self.fill_form_task(url, form_data, submit)

from pydantic import BaseModel, Field
from typing import Literal, Optional

class FlightDetails(BaseModel):
    """Flight booking details"""
    origin: str = Field(description="Departure city/airport code")
    destination: str = Field(description="Arrival city/airport code")
    departure_date: str = Field(description="Departure date (YYYY-MM-DD)")
    return_date: Optional[str] = Field(default=None, description="Return date for round-trip (YYYY-MM-DD)")
    passengers: int = Field(default=1, description="Number of passengers")
    class_type: Literal["economy", "business", "first"] = Field(default="economy")
    budget: Optional[float] = Field(default=None, description="Maximum budget per person in USD")

class HotelDetails(BaseModel):
    """Hotel booking details"""

    check_in: str = Field(description="Check-in date (YYYY-MM-DD)")

    rooms: int = Field(default=1, description="Number of rooms")

        """Compare the same item across multiple websites"""

        Compare {item_name} across these websites: {', '.join(websites)}
        Focus on these criteria: {', '.join(criteria)}

        - Price (including any additional fees)
        - Rating/reviews

        - Availability and shipping

            extracted_content=f"Comparison task initiated for {item_name} across {len(websites)} websites",
            include_in_memory=True
        )

        items: List[dict],
        criteria: List[str]

            comparison_items = [ComparisonItem(**item) for item in items]

            # Find best quality (highest rating)

            best_quality = max(rated_items, key=lambda x: x.rating) if rated_items else None
            
            summary = f"""
            COMPARISON SUMMARY
            ===================
            Total Items Compared: {len(comparison_items)}
            Criteria Used: {', '.join(criteria)}
            
            BEST VALUE: {best_value.name} at {best_value.provider} - ${best_value.price:.2f}
            """
            
            if best_quality:
                summary += f"\nBEST QUALITY: {best_quality.name} at {best_quality.provider} - Rating: {best_quality.rating}/5"
            
            summary += "\n\nDETAILED COMPARISON:"
            for i, item in enumerate(comparison_items, 1):
                summary += f"\n{i}. {item.name} ({item.provider})"
                summary += f"\n   Price: ${item.price:.2f}"
                if item.rating:
                    summary += f"\n   Rating: {item.rating}/5"
                if item.features:
                    summary += f"\n   Features: {', '.join(item.features[:3])}"
            
            logger.info("Comparison summary generated successfully")
            
            return ActionResult(
                extracted_content=summary,
                include_in_memory=True

            return ActionResult(error=f"Failed to generate summary: {str(e)}")

        return ActionResult(
            extracted_content=ranking_summary,
            include_in_memory=True
        )

import logging
import asyncio
from functools import wraps
from typing import Callable, Any

logger = logging.getLogger(__name__)

def handle_errors(max_retries: int = 3):
    """Decorator for error handling with retries and exponential backoff"""
    def decorator(func: Callable) -> Callable:
        @wraps(func)
        async def wrapper(*args, **kwargs) -> Any:
            last_error = None
            for attempt in range(max_retries):
                try:
                    return await func(*args, **kwargs)
                except Exception as e:
                    last_error = e
                    logger.warning(f"Attempt {attempt + 1}/{max_retries} failed for {func.__name__}: {str(e)}")
                    if attempt < max_retries - 1:
                        backoff_time = 2 ** attempt
                        logger.info(f"Retrying in {backoff_time} seconds...")
                        await asyncio.sleep(backoff_time)
            logger.error(f"All {max_retries} attempts failed for {func.__name__}")
            raise last_error
        return wrapper
    return decorator

import os
import re
from typing import Dict
from urllib.parse import urlparse
import logging

logger = logging.getLogger(__name__)

def sanitize_sensitive_data(data: Dict) -> Dict:
    """Remove sensitive data from logs and outputs"""
    sensitive_keys = ['password', 'api_key', 'token', 'credit_card', 'ssn', 'secret']
    sanitized = data.copy()
    
    for key in list(sanitized.keys()):
        if any(sensitive in key.lower() for sensitive in sensitive_keys):
            sanitized[key] = "***REDACTED***"
            logger.debug(f"Redacted sensitive field: {key}")
    
    return sanitized

def validate_url(url: str, allowed_domains: list[str] | None = None) -> bool:
    """Validate URL is safe and allowed"""
    try:
        parsed = urlparse(url)
        
        # Basic URL validation
        if not all([parsed.scheme, parsed.netloc]):
            return False
        
        if parsed.scheme not in ['http', 'https']:
            return False
        
        # Domain validation if allowed domains provided
        if allowed_domains:
            domain = parsed.netloc.replace('www.', '')
            if not any(allowed in domain for allowed in allowed_domains):
                logger.warning(f"URL domain not in allowed list: {domain}")
                return False
        
        return True
    except Exception as e:
        logger.error(f"URL validation error: {e}")
        return False

def sanitize_user_input(input_text: str) -> str:
    """Sanitize user input to prevent injection attacks"""
    # Remove potential dangerous characters
    sanitized = re.sub(r'[<>"\']', '', input_text)
    return sanitized.strip()

def async_retry(max_attempts: int = 3, delay: float = 1.0):
    """Async retry decorator"""
    def decorator(func):
        async def wrapper(*args, **kwargs):
            for attempt in range(max_attempts):
                try:
                    return await func(*args, **kwargs)
                except Exception as e:
                    if attempt == max_attempts - 1:
                        raise
                    logger.warning(f"Attempt {attempt + 1} failed, retrying in {delay}s: {e}")
                    await asyncio.sleep(delay)
        return wrapper
    return decorator

14. TROUBLESHOOTING
================================================================================

COMMON ISSUES AND SOLUTIONS:
----------------------------

1. INSTALLATION ISSUES
   Problem: "Module not found" errors
   Solution: Ensure virtual environment is activated and dependencies installed
   ```bash
   source venv/bin/activate
   pip install -r requirements.txt
   ```

2. API KEY ISSUES
   Problem: Authentication errors
   Solution: Check .env file and ensure BROWSER_USE_API_KEY is set correctly
   ```bash
   echo $BROWSER_USE_API_KEY  # Should show your API key
   ```

3. BROWSER LAUNCH ISSUES
   Problem: Browser fails to start
   Solution: Set headless=False for debugging, check system dependencies
   ```python
   agent = Agent(task="...", headless=False)
   ```

4. TIMEOUT ISSUES
   Problem: Actions timing out
   Solution: Increase timeout values in settings
   ```python
   # In config/settings.py
   step_timeout = 300  # Increase from 180 to 300 seconds
   ```

5. MEMORY ISSUES
   Problem: High memory usage
   Solution: Use max_history_items to limit memory
   ```python
   agent = Agent(task="...", max_history_items=50)
   ```

6. FORM FILLING FAILURES
   Problem: Agent can't find form elements
   Solution: Use more specific task descriptions or custom selectors
   ```python
   task = "Fill the form with ID 'contact-form' using the provided data"
   ```

7. BOOKING SITE CHANGES
   Problem: Booking sites updated their UI
   Solution: Agent will adapt automatically, but may need task description updates

8. COMPARISON INACCURACIES
   Problem: Comparison results seem incorrect
   Solution: Verify criteria are specific and websites are accessible

DEBUGGING TIPS:
---------------
1. Enable detailed logging
   ```python
   import logging
   logging.basicConfig(level=logging.DEBUG)
   ```

2. Use headless=False to see browser actions
   ```python
   agent = Agent(task="...", headless=False)
   ```

3. Check agent history for detailed steps
   ```python
   history = await agent.run()
   print(history.action_names())  # See all actions taken
   print(history.errors())  # See any errors
   ```

4. Test with simple tasks first
   ```python
   # Start with simple navigation
   agent = Agent(task="Go to google.com and tell me the page title")
   ```
